Redux Auth State Shape

// store.auth — initial state { user: null, // { id, name, email, tenant } token: null, // JWT string role: null, // 'user' | 'admin' | 'super_admin' isAuthenticated: false, isLoading: false, error: null, } // Actions — import from features/auth/authSlice.js setCredentials({ user, token, role }) // call after login success logout() // clears all auth state setLoading(bool) // toggle loading setError(string) // store error message clearError() // reset error // Selectors — use with useSelector() selectCurrentUser // → user object selectIsAuthenticated // → boolean selectUserRole // → 'user' | 'admin' | 'super_admin' selectAuthToken // → JWT string selectAuthLoading // → boolean selectAuthError // → string | null
